/* ---------------- shared helpers ---------------- */
function fm() { return FileManager.local(); }
function path(name) { return fm().joinPath(fm().documentsDirectory(), name); }
function readJSON(p) { try { if (!fm().fileExists(p)) return null; return JSON.parse(fm().readString(p)); } catch (e) { return null; } }
function writeJSON(p, o) { try { fm().writeString(p, JSON.stringify(o)); } catch (e) {} }
function dropFile(p) { try { if (fm().fileExists(p)) fm().remove(p); } catch (e) {} }

function call(url, payload) {
  var req = new Request(url);
  req.timeoutInterval = 20;
  req.headers = { "Accept": "application/json" };
  if (payload) {
    req.method = "POST";
    req.headers["Content-Type"] = "application/json";
    req.body = JSON.stringify(payload);
  }
  return req.loadJSON();
}

function sign(n) {
  if (n === null || n === undefined || isNaN(n)) return "-";
  if (n === 0) return "E";
  return (n > 0 ? "+" : "") + n;
}
function today() {
  var d = new Date(), m = d.getMonth() + 1, day = d.getDate();
  return d.getFullYear() + "-" + (m < 10 ? "0" + m : m) + "-" + (day < 10 ? "0" + day : day);
}
function shortDate(iso) {
  if (!iso) return "";
  var p = String(iso).split("-");
  if (p.length !== 3) return String(iso);
  var M = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  return (M[parseInt(p[1], 10) - 1] || p[1]) + " " + parseInt(p[2], 10);
}
// Pull the hole-by-hole scores and pars out of a saved round's notes.
function parseNotes(r) {
  if (!r || !r.notes) return null;
  var m = String(r.notes).match(/Hole by hole:\s*([0-9]+(?:\/[0-9]+)*)\s*\(\s*par\s*([0-9]+(?:\/[0-9]+)*)/i);
  if (!m) return null;
  var s = m[1].split("/").map(function (x) { return parseInt(x, 10); });
  var p = m[2].split("/").map(function (x) { return parseInt(x, 10); });
  if (!s.length || !p.length || s.length !== p.length) return null;
  for (var i = 0; i < s.length; i++) { if (isNaN(s[i]) || isNaN(p[i])) return null; }
  var holes = null;
  var hm = String(r.notes).match(/\[holes\s+([0-9]+(?:[ ,]+[0-9]+)*)\]/i);
  if (hm) {
    holes = hm[1].split(/[ ,]+/).map(function (x) { return parseInt(x, 10); });
    if (holes.length !== s.length) holes = null;
  }
  return { scores: s, pars: p, holes: holes };
}
function sum(a) { var t = 0; for (var i = 0; i < a.length; i++) t += (a[i] || 0); return t; }
function sumSlice(a, from, to) { var t = 0; for (var i = from; i < to && i < a.length; i++) t += (a[i] || 0); return t; }

function getData() {
  try {
    var d = call(ROUNDS);
    if (d && d.ok && d.stats) { writeJSON(path(CACHE), d); return { data: d, stale: false }; }
  } catch (e) {}
  var c = readJSON(path(CACHE));
  if (c && c.stats) return { data: c, stale: true };
  return { data: null, stale: false };
}

function wp(st, str, size, color, bold, center) {
  var t = st.addText(String(str));
  t.font = bold ? Font.boldSystemFont(size) : Font.systemFont(size);
  t.textColor = color;
  t.lineLimit = 1;
  t.minimumScaleFactor = 0.7;
  if (center) t.centerAlignText();
  return t;
}
function chip(st, label, fg, bg, size, url, pad) {
  var c = st.addStack();
  c.backgroundColor = bg;
  c.cornerRadius = 5;
  c.setPadding(2, pad || 5, 2, pad || 5);
  c.centerAlignContent();
  var t = wp(c, label, size, fg, true, true);
  if (url) c.url = url;
  return c;
}
function selfURL(extra) {
  return "scriptable:///run/" + encodeURIComponent(SELF) + (extra ? "?" + extra : "");
}

// The draft is the round in progress: what the widget edits and the card finishes.
function blankDraft(data) {
  var pars = [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4];
  var course = (data && data.stats && data.stats.lastCourse) || "Ala Wai";
  var island = "Oahu";
  var courses = (data && data.courses) ? data.courses : [];
  for (var i = 0; i < courses.length; i++) if (courses[i].name === course) island = courses[i].island || island;
  var rounds = (data && data.rounds) ? data.rounds : [];
  for (var r = 0; r < rounds.length; r++) {
    var pn = parseNotes(rounds[r]);
    if (rounds[r].course === course && pn) { pars = pn.pars.concat(pars.slice(pn.pars.length)); break; }
  }
  return { course: course, island: island, holes: 18, pars: pars, scores: [], preview: true };
}

function buildCard(data) {
  var w = new ListWidget();
  // paper scorecard look
  var bg = new Color("#f7f2e3"), ink = new Color("#14321f"), soft = new Color("#6b7a63");
  var green = new Color("#1f7a4a"), gold = new Color("#f0d97a"), softBtn = new Color("#e2ddc8");
  w.backgroundColor = bg;
  w.setPadding(12, 14, 10, 14);

  var st = data ? data.stats : null;
  var rounds = (data && data.rounds) ? data.rounds : [];
  var draft = readJSON(path(DRAFT)) || blankDraft(data);
  if (!draft.scores) draft.scores = [];
  if (!draft.pars) draft.pars = [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4];

  // No round in progress: mirror the last saved card.
  var preview = !!draft.preview && !(draft.scores && draft.scores.length);
  var shownScores = draft.scores || [];
  var courseName = draft.course;
  if (preview && rounds.length && rounds[0].course) {
    var r0 = rounds[0];
    var pn = parseNotes(r0);
    var hs = (r0.holeScores && r0.holeScores.length) ? r0.holeScores : (pn ? pn.scores : null);
    if (hs && hs.length) {
      if (pn && pn.holes) {
        var mapped = [];
        for (var mi = 0; mi < pn.holes.length; mi++) mapped[pn.holes[mi] - 1] = hs[mi];
        shownScores = mapped;
      } else {
        shownScores = hs;
      }
      courseName = r0.course;
      if (pn && pn.pars) { var pp = draft.pars.slice(0); for (var pi = 0; pi < pn.pars.length && pi < 18; pi++) pp[pi] = pn.pars[pi]; draft.pars = pp; }
    }
  }

  var hole = Number(draft.hole) || 1;
  if (hole < 1) hole = 1;
  if (hole > 18) hole = 18;
  var idx = hole - 1;
  var par = draft.pars[idx] || 4;
  var score = (shownScores[idx] === undefined || shownScores[idx] === null) ? null : shownScores[idx];

  var tot = 0, parPlayed = 0, played = 0;
  for (var i = 0; i < 18; i++) {
    var sv = shownScores[i];
    if (sv === undefined || sv === null) continue;
    tot += sv; parPlayed += (draft.pars[i] || 4); played++;
  }

  // row 1: title + course count
  var r1 = w.addStack(); r1.layoutHorizontally(); r1.centerAlignContent();
  wp(r1, "HAWAII GOLF", 11, ink, true);
  r1.addSpacer();
  if (st) wp(r1, st.coursesPlayed + "/" + st.coursesTotal + " HI", 8.5, soft, true);
  w.addSpacer(5);

  // row 2: course + standing
  var r2 = w.addStack(); r2.layoutHorizontally(); r2.centerAlignContent();
  var ct = wp(r2, courseName || "Pick a course", 11, ink, true);
  ct.lineLimit = 1; ct.minimumScaleFactor = 0.6;
  r2.addSpacer();
  wp(r2, played ? (tot + "  " + sign(tot - parPlayed)) : "no scores yet", 11, played ? green : soft, true);
  w.addSpacer(6);

  // row 3: hole + par
  var r3 = w.addStack(); r3.layoutHorizontally(); r3.centerAlignContent();
  wp(r3, "HOLE " + hole + " OF 18", 9, soft, true);
  r3.addSpacer();
  wp(r3, "PAR " + par, 9, soft, true);
  w.addSpacer(3);

  // row 4: the hole, with its arrows
  var big = w.addStack(); big.layoutHorizontally(); big.centerAlignContent();
  var minus = big.addStack();
  minus.backgroundColor = softBtn; minus.cornerRadius = 12;
  minus.size = new Size(52, 42); minus.centerAlignContent();
  wp(minus, "-", 26, ink, true, true);
  minus.url = selfURL("bump=" + hole + "&d=-1&hole=" + hole);
  big.addSpacer();
  var mid = big.addStack(); mid.layoutVertically(); mid.centerAlignContent();
  wp(mid, score === null ? "-" : String(score), 40, score === null ? soft : ink, true, true);
  wp(mid, score === null ? "tap to score" : (score === par ? "even par" : (score < par ? (par - score) + " under" : (score - par) + " over")), 9, soft, false, true);
  big.addSpacer();
  var plus = big.addStack();
  plus.backgroundColor = green; plus.cornerRadius = 12;
  plus.size = new Size(52, 42); plus.centerAlignContent();
  wp(plus, "+", 26, Color.white(), true, true);
  plus.url = selfURL("bump=" + hole + "&d=1&hole=" + hole);
  w.addSpacer(7);

  // row 5: nav + save + full card
  var r5 = w.addStack(); r5.layoutHorizontally(); r5.centerAlignContent(); r5.spacing = 6;
  chip(r5, "<", ink, softBtn, 15, selfURL("hole=" + (hole > 1 ? hole - 1 : 18)), 10);
  chip(r5, ">", ink, softBtn, 15, selfURL("hole=" + (hole < 18 ? hole + 1 : 1)), 10);
  r5.addSpacer();
  chip(r5, "SAVE", Color.white(), green, 9, selfURL("save=1"), 9);
  chip(r5, "FULL CARD", ink, gold, 9, selfURL("card=1"), 9);
  w.addSpacer(5);

  // row 6: island progress + next course
  var r6 = w.addStack(); r6.layoutHorizontally(); r6.centerAlignContent();
  var ip = (data && data.islandProgress) ? data.islandProgress : null;
  var islandTxt = "";
  if (ip) {
    var keys = Object.keys(ip);
    for (var z = 0; z < keys.length; z++) {
      if (ip[keys[z]].played < ip[keys[z]].total) { islandTxt = keys[z].toUpperCase() + " " + ip[keys[z]].played + "/" + ip[keys[z]].total; break; }
    }
  }
  if (islandTxt) wp(r6, islandTxt, 8, green, true);
  r6.addSpacer();
  var nextName = "";
  var courses = (data && data.courses) ? data.courses : [];
  var playedList = (data && data.coursesPlayedList) ? data.coursesPlayedList : [];
  for (var n = 0; n < courses.length; n++) if (playedList.indexOf(courses[n].name) === -1) { nextName = courses[n].name; break; }
  wp(r6, nextName ? ("NEXT " + nextName) : "ALL 73 PLAYED", 8, soft, false);

  w.url = selfURL("card=1");
  w.refreshAfterDate = new Date(Date.now() + 60 * 1000);
  return w;
}

function renderWidget() {
  var got = getData();
  var w = null;
  try {
    w = buildCard(got.data);
  } catch (e) {
    w = new ListWidget();
    w.backgroundColor = new Color("#08301c");
    w.setPadding(12, 12, 12, 12);
    var t = w.addText("HAWAII GOLF");
    t.font = Font.boldSystemFont(11);
    t.textColor = new Color("#f3e6bd");
    var e2 = w.addText(String(e).slice(0, 140));
    e2.font = Font.systemFont(9);
    e2.textColor = Color.white();
    w.url = selfURL("");
  }
  Script.setWidget(w);
  Script.complete();
}

function defaultPars() { return [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4]; }

function loadDraft(data) {
  var d = readJSON(path(DRAFT));
  if (!d || !d.course) d = blankDraft(data);
  if (!d.scores) d.scores = [];
  if (!d.pars) d.pars = defaultPars();
  if (!d.holes) d.holes = 18;
  return d;
}
function saveDraft(d) { writeJSON(path(DRAFT), d); }
function hasScore(d, i) { return d.scores[i] !== undefined && d.scores[i] !== null; }
function scoredCount(d) { var n = 0; for (var i = 0; i < 18; i++) if (hasScore(d, i)) n++; return n; }
function roundTotals(d) {
  var s = 0, p = 0, n = 0;
  for (var i = 0; i < 18; i++) {
    if (!hasScore(d, i)) continue;
    s += d.scores[i]; p += (d.pars[i] || 4); n++;
  }
  return { score: s, par: p, holes: n };
}

function infoRow(t, title, subtitle, color) {
  var r = new UITableRow(); r.height = subtitle ? 62 : 46;
  var c = r.addText(String(title), subtitle ? String(subtitle) : null);
  c.widthWeight = 100;
  if (color) c.titleColor = color;
  t.addRow(r);
  return r;
}
function actionRow(t, title, subtitle, tap, color) {
  var r = new UITableRow(); r.height = subtitle ? 62 : 52;
  r.dismissOnSelect = false;
  var c = r.addText(String(title), subtitle ? String(subtitle) : null);
  c.widthWeight = 100;
  if (color) c.titleColor = color;
  if (tap) r.onSelect = tap;
  t.addRow(r);
  return r;
}
function holeRow(t, draft, i, tap) {
  var par = draft.pars[i] || 4;
  var sv = hasScore(draft, i) ? String(draft.scores[i]) : "-";
  var r = new UITableRow(); r.height = 54; r.dismissOnSelect = false;
  var left = r.addText("Hole " + (i + 1), "Par " + par);
  left.widthWeight = 58;
  var mid = r.addText(hasScore(draft, i) ? sign(draft.scores[i] - par) : "", null);
  mid.widthWeight = 18; mid.centerAligned = true;
  if (hasScore(draft, i)) mid.titleColor = draft.scores[i] < par ? Color.green() : (draft.scores[i] > par ? Color.red() : Color.gray());
  var right = r.addText(sv, null);
  right.widthWeight = 24; right.centerAligned = true;
  right.titleFont = Font.boldSystemFont(20);
  r.onSelect = tap;
  t.addRow(r);
  return { row: r, mid: mid, right: right };
}

async function pickCourse(draft, data, refresh) {
  var t = new UITable();
  infoRow(t, "CHOOSE A COURSE", "Tap one, or type any course in the world");
  actionRow(t, "Type any course in the world", "Free text, any country", async function () {
    var a = new Alert();
    a.title = "Course";
    a.addTextField("Course name", draft.course || "");
    a.addTextField("City, state or island", draft.island || "");
    a.addAction("Use this course");
    a.addCancelAction("Cancel");
    var k = await a.presentAlert();
    if (k !== 0) return;
    var cn = a.textFieldValue(0).trim();
    if (!cn) return;
    draft.course = cn;
    draft.island = a.textFieldValue(1).trim() || "Other";
    saveDraft(draft);
    await loadTemplatePars(draft);
    if (refresh) await refresh();
    t.reload();
  });
  var courses = (data && data.courses) ? data.courses : [];
  for (var i = 0; i < courses.length; i++) {
    (function (c) {
      actionRow(t, c.name, c.island, async function () {
        draft.course = c.name;
        draft.island = c.island || "Oahu";
        saveDraft(draft);
        await loadTemplatePars(draft);
        if (refresh) await refresh();
        t.reload();
      });
    })(courses[i]);
  }
  actionRow(t, "Close", null, async function () { t.reload(); });
  await t.present();
}

async function loadTemplatePars(draft) {
  try {
    var url = TEMPLATES + "?course=" + encodeURIComponent(draft.course);
    var res = call(url);
    var p = res && res.template && res.template.pars;
    if (p && p.length) {
      for (var i = 0; i < p.length && i < 18; i++) draft.pars[i] = p[i];
      saveDraft(draft);
    }
  } catch (e) {}
}

async function setOther(draft, i, init, reload) {
  var b = new Alert();
  b.title = "Hole " + (i + 1) + " score";
  b.message = "Strokes taken";
  b.addTextField("Strokes", String(init));
  b.addAction("Set");
  b.addCancelAction("Cancel");
  var j = await b.presentAlert();
  if (j !== 0) return;
  var v = parseInt(b.textFieldValue(0), 10);
  if (!isNaN(v) && v > 0 && v < 21) {
    draft.scores[i] = v;
    draft.preview = false;
    saveDraft(draft);
  }
  await reload();
}

async function holePrompt(draft, i, reload) {
  var par = draft.pars[i] || 4;
  var cur = hasScore(draft, i) ? draft.scores[i] : null;
  var a = new Alert();
  a.title = "Hole " + (i + 1) + "   par " + par;
  var picks = [];
  if (cur === null) {
    a.message = "Tap your score. Every tap saves it.";
    picks = [
      { label: "Par  " + par, v: par },
      { label: (par - 1) + "  birdie", v: par - 1 },
      { label: (par + 1) + "  bogey", v: par + 1 },
      { label: (par + 2) + "  double", v: par + 2 }
    ];
    for (var x = 0; x < picks.length; x++) a.addAction(picks[x].label);
    a.addAction("Other number");
  } else {
    a.message = "Now " + cur + "  (" + sign(cur - par) + ")";
    a.addAction("+1   to " + (cur + 1));
    a.addAction("-1   to " + Math.max(1, cur - 1));
    a.addAction("Other number");
    a.addAction("Clear this hole");
  }
  a.addAction("Change par  (now " + par + ")");
  a.addCancelAction("Cancel");

  var k = await a.presentAlert();
  if (k === -1) return;

  var scoreActions = (cur === null) ? picks.length + 1 : 4;
  if (k === scoreActions) {
    var order = [3, 4, 5, 6];
    draft.pars[i] = order[(order.indexOf(par) + 1) % 4];
    saveDraft(draft);
    await reload();
    return;
  }

  draft.preview = false;
  if (cur === null) {
    if (k < picks.length) draft.scores[i] = picks[k].v;
    else if (k === picks.length) { await setOther(draft, i, par, reload); return; }
  } else {
    if (k === 0) draft.scores[i] = cur + 1;
    else if (k === 1) draft.scores[i] = Math.max(1, cur - 1);
    else if (k === 2) { await setOther(draft, i, cur, reload); return; }
    else if (k === 3) draft.scores[i] = null;
  }
  saveDraft(draft);
  await reload();
}

async function saveRound(draft) {
  var used = [], usedPars = [], usedHoles = [];
  var last = (draft.holes === 9) ? 9 : 18;
  for (var i = 0; i < last; i++) {
    if (!hasScore(draft, i)) continue;
    used.push(draft.scores[i]);
    usedPars.push(draft.pars[i] || 4);
    usedHoles.push(i + 1);
  }
  if (!used.length) {
    var w = new Alert();
    w.title = "Nothing to save";
    w.message = "Tap a hole and put a score in first.";
    w.addAction("OK");
    await w.presentAlert();
    return false;
  }
  var body = {
    action: "create",
    course: draft.course,
    island: draft.island || "Other",
    date: today(),
    holeScores: used,
    parsUsed: sum(usedPars),
    score: sum(used),
    holesPlayed: used.length,
    notes: "Hole by hole: " + used.join("/") + " (par " + usedPars.join("/") + ") [holes " + usedHoles.join(",") + "]"
  };
  var res = null;
  try { res = call(ROUNDS, body); } catch (e) { res = null; }
  var ok = res && res.ok;
  var a = new Alert();
  a.title = ok ? "Round saved" : "Could not save";
  a.message = ok
    ? (used.length + " holes at " + draft.course + " sent to the Daily Dashboard.")
    : ((res && (res.error || res.dashError)) || "No response from the server.");
  a.addAction("OK");
  await a.presentAlert();
  if (!ok) return false;
  draft.scores = [];
  draft.preview = false;
  saveDraft(draft);
  dropFile(path(CACHE));
  return true;
}

async function roundsList() {
  var d = call(ROUNDS);
  var rounds = (d && d.rounds) ? d.rounds : [];
  var t = new UITable();
  infoRow(t, "SAVED ROUNDS", rounds.length + " round" + (rounds.length === 1 ? "" : "s") + " - tap one to delete it");
  if (!rounds.length) infoRow(t, "No rounds yet", "Nothing saved so far.");
  for (var i = 0; i < rounds.length; i++) {
    (function (r) {
      var row = new UITableRow(); row.height = 62; row.dismissOnSelect = false;
      var l = row.addText(r.course + (r.island ? "  |  " + r.island : ""), shortDate(r.date) + "  |  " + r.holes + " holes  |  par " + r.par);
      l.widthWeight = 72;
      var rt = row.addText(String(r.score), r.toPar === 0 ? "E" : sign(r.toPar));
      rt.widthWeight = 28; rt.centerAligned = true;
      rt.titleFont = Font.boldSystemFont(20);
      row.onSelect = async function () {
        var a = new Alert();
        a.title = "Delete this round?";
        a.message = r.course + "  |  " + shortDate(r.date) + "  |  " + r.score + " in " + r.holes + " holes";
        a.addDestructiveAction("Delete round");
        a.addCancelAction("Keep it");
        var k = await a.presentAlert();
        if (k !== 0) return;
        var res = null;
        try { res = call(ROUNDS, { action: "delete", id: r.id, course: r.course, date: r.date, score: r.score }); } catch (e) {}
        dropFile(path(CACHE));
        var done = new Alert();
        done.title = (res && res.ok) ? "Round deleted" : "Could not delete";
        done.message = (res && res.ok) ? (r.course + " removed.") : ((res && res.error) || "Server did not answer.");
        done.addAction("OK");
        await done.presentAlert();
        t.reload();
      };
      t.addRow(row);
    })(rounds[i]);
  }
  actionRow(t, "Close", null, async function () { t.reload(); });
  await t.present();
}

async function scorecard(draft, data) {
  var t = new UITable();
  var slots = [];
  var headCell = null;

  function refreshHeader() {
    var tot = roundTotals(draft);
    if (!headCell) return;
    headCell.title = draft.course + (draft.island && draft.island !== "Oahu" ? "  |  " + draft.island : "");
    headCell.subtitle = tot.holes
      ? (tot.holes + " holes in, " + tot.score + "  (" + sign(tot.score - tot.par) + ")")
      : "No scores yet - tap a hole below";
  }
  function refreshHoles() {
    for (var i = 0; i < slots.length; i++) {
      var s = slots[i];
      if (!s) continue;
      var par = draft.pars[i] || 4;
      if (hasScore(draft, i)) {
        s.right.title = String(draft.scores[i]);
        s.right.titleColor = null;
        s.mid.title = sign(draft.scores[i] - par);
        s.mid.titleColor = draft.scores[i] < par ? Color.green() : (draft.scores[i] > par ? Color.red() : Color.gray());
      } else {
        s.right.title = "-";
        s.mid.title = "";
      }
    }
  }
  async function refresh() {
    refreshHeader();
    refreshHoles();
    await t.reload();
  }

  var head = new UITableRow(); head.height = 74;
  headCell = head.addText(draft.course, "No scores yet - tap a hole below");
  headCell.widthWeight = 100;
  t.addRow(head);
  refreshHeader();

  var last = (draft.holes === 9) ? 9 : 18;
  for (var i = 0; i < last; i++) {
    (function (idx) {
      var made = holeRow(t, draft, idx, async function () {
        await holePrompt(draft, idx, refresh);
      });
      slots[idx] = made;
    })(i);
  }

  actionRow(t, "SAVE ROUND", "Sends only the holes you scored, to the Daily Dashboard", async function () {
    var saved = await saveRound(draft);
    if (saved) await refresh();
  }, Color.blue());

  actionRow(t, "Course and par card", draft.course, async function () {
    await pickCourse(draft, data, null);
    refresh();
  });

  actionRow(t, "Holes", (draft.holes === 9 ? "Front 9" : "Full 18"), async function () {
    draft.holes = (draft.holes === 9) ? 18 : 9;
    saveDraft(draft);
    t.reload();
  });

  actionRow(t, "Clear all scores", "Wipes the round in progress, saved rounds stay", async function () {
    var a = new Alert();
    a.title = "Clear the round?";
    a.message = "Scores in progress are erased. Rounds already saved are kept.";
    a.addDestructiveAction("Clear");
    a.addCancelAction("Cancel");
    var k = await a.presentAlert();
    if (k !== 0) return;
    draft.scores = [];
    saveDraft(draft);
    await refresh();
  }, Color.red());

  actionRow(t, "Main menu", null, async function () {
    await mainMenu(loadDraft(data), data);
  });

  await t.present();
}

async function mainMenu(draft, data) {
  var t = new UITable();
  var stats = (data && data.stats) ? data.stats : null;
  var tot = roundTotals(draft);

  var head = new UITableRow(); head.height = 74;
  var ht = head.addText("HAWAII GOLF", stats ? (stats.coursesPlayed + " of " + stats.coursesTotal + " Hawaii courses played" + (stats.worldPlayed ? " | " + stats.worldPlayed + " worldwide" : "")) : "Play them all");
  ht.widthWeight = 100;
  t.addRow(head);

  actionRow(t, tot.holes ? "Continue round" : "Start a round", tot.holes
    ? (draft.course + "  |  " + tot.holes + " holes in, " + tot.score + " (" + sign(tot.score - tot.par) + ")")
    : (draft.course + "  |  par card ready"), async function () {
      var d2 = loadDraft(data);
      await scorecard(d2, data);
    }, Color.blue());

  actionRow(t, "Course", draft.course + (draft.island ? "  |  " + draft.island : ""), async function () {
    await pickCourse(draft, data, null);
    t.reload();
  });

  actionRow(t, "Holes", (draft.holes === 9 ? "Front 9" : "Full 18"), async function () {
    draft.holes = (draft.holes === 9) ? 18 : 9;
    saveDraft(draft);
    await t.reload();
  });

  actionRow(t, "Saved rounds", stats ? (stats.totalRounds + " saved" + (stats.lastCourse ? " | last " + stats.lastCourse + " " + stats.lastScore : "")) : "Nothing saved yet", async function () {
    await roundsList();
  });

  if (stats) {
    infoRow(t, "Island progress", (function () {
      var ip = (data && data.islandProgress) ? data.islandProgress : {};
      var parts = [];
      var keys = Object.keys(ip);
      for (var i = 0; i < keys.length; i++) parts.push(keys[i] + " " + ip[keys[i]].played + "/" + ip[keys[i]].total);
      return parts.join("   ");
    })());
    if (stats.averageScore) infoRow(t, "Average", stats.averageScore + " per round" + (stats.bestCourse ? "  |  best " + stats.bestCourse + " " + sign(stats.bestToPar) : ""));
  }

  await t.present();
}

/* ---------------- widget taps (URL scheme) ---------------- */
var query = {};
try { if (typeof args !== "undefined" && args.queryParameters) query = args.queryParameters; } catch (e) { query = {}; }

function wipDraft() {
  var d = readJSON(path(DRAFT));
  if (!d || !d.course) { var got = getData(); d = blankDraft(got.data); }
  if (!d.scores) d.scores = [];
  if (!d.pars) d.pars = [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4];
  return d;
}

function applyBump(hole, delta, keepHole) {
  var d = wipDraft();
  if (d.preview) { d.scores = []; d.preview = false; }
  var i = Number(hole) - 1;
  if (isNaN(i) || i < 0 || i > 17) return d;
  var par = d.pars[i] || 4;
  var set = !(d.scores[i] === undefined || d.scores[i] === null);
  var next = set ? d.scores[i] + Number(delta || 1) : par;
  if (next < 1) next = 1;
  if (next > 20) next = 20;
  d.scores[i] = next;
  if (keepHole) d.hole = Math.min(18, Math.max(1, Number(keepHole)));
  writeJSON(path(DRAFT), d);
  return d;
}

function saveFromWidget() {
  var d = wipDraft();
  var scores = d.scores || [];
  var pars = d.pars || [];
  var used = [], usedPars = [], usedHoles = [];
  for (var hi = 0; hi < 18; hi++) {
    var sv = scores[hi];
    if (sv === undefined || sv === null) continue;
    used.push(sv); usedPars.push(pars[hi] || 4); usedHoles.push(hi + 1);
  }
  if (!used.length) return Script.complete();
  var body = {
    action: "create",
    course: d.course,
    island: d.island || "Other",
    holeScores: used,
    date: today(),
    score: sum(used),
    parsUsed: sum(usedPars),
    holesPlayed: used.length,
    notes: "Hole by hole: " + used.join("/") + " (par " + usedPars.join("/") + ") [holes " + usedHoles.join(",") + "]"
  };
  var res = null;
  try { res = call(ROUNDS, body); } catch (e) {}
  var a = new Alert();
  a.title = (res && res.ok) ? "Round saved" : "Could not save";
  a.message = (res && res.ok)
    ? (used.length + " holes at " + d.course + " sent to the Daily Dashboard.")
    : ((res && (res.error || res.dashError)) || "No answer from the server.");
  a.addAction("OK");
  if (res && res.ok) {
    d.scores = [];
    d.preview = false;
    writeJSON(path(DRAFT), d);
    dropFile(path(CACHE));
  }
  return a.present().then(function () { Script.complete(); });
}

async function run() {
  var got = getData();
  var data = got.data;
  var draft = loadDraft(data);
  if (query && query.bump) {
    applyBump(query.bump, query.d, query.hole);
    return scorecard(loadDraft(data), data);
  }
  if (query && query.hole) {
    var hd = loadDraft(data);
    hd.hole = Math.min(18, Math.max(1, Number(query.hole)));
    saveDraft(hd);
    return scorecard(hd, data);
  }
  if (query && (query.card || query.open)) return scorecard(draft, data);
  return mainMenu(draft, data);
}

/* ---------------- entry ---------------- */
if (isWidget) {
  renderWidget();
} else if (query.save) {
  saveFromWidget();
} else {
  run().then(function () { Script.complete(); }).catch(function (e) {
    var a = new Alert();
    a.title = "Hawaii Golf error";
    a.message = String(e).slice(0, 220);
    a.addAction("OK");
    a.presentAlert().then(function () { Script.complete(); });
  });
}
