/* ---------------- shared helpers ---------------- */
function fm() { return FileManager.local(); }
function path(name) { return fm().joinPath(fm().documentsDirectory(), name); }
function readJSON(p) { try { if (!fm().fileExists(p)) return null; return JSON.parse(fm().readString(p)); } catch (e) { return null; } }
function writeJSON(p, o) { try { fm().writeString(p, JSON.stringify(o)); } catch (e) {} }
function dropFile(p) { try { if (fm().fileExists(p)) fm().remove(p); } catch (e) {} }

function call(url, payload) {
  var req = new Request(url);
  req.timeoutInterval = 20;
  req.headers = { "Accept": "application/json" };
  if (payload) {
    req.method = "POST";
    req.headers["Content-Type"] = "application/json";
    req.body = JSON.stringify(payload);
  }
  return req.loadJSON();
}

function sign(n) {
  if (n === null || n === undefined || isNaN(n)) return "-";
  if (n === 0) return "E";
  return (n > 0 ? "+" : "") + n;
}
function today() {
  var d = new Date(), m = d.getMonth() + 1, day = d.getDate();
  return d.getFullYear() + "-" + (m < 10 ? "0" + m : m) + "-" + (day < 10 ? "0" + day : day);
}
function shortDate(iso) {
  if (!iso) return "";
  var p = String(iso).split("-");
  if (p.length !== 3) return String(iso);
  var M = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  return (M[parseInt(p[1], 10) - 1] || p[1]) + " " + parseInt(p[2], 10);
}
// Pull the hole-by-hole scores and pars out of a saved round's notes.
function parseNotes(r) {
  if (!r || !r.notes) return null;
  var m = String(r.notes).match(/Hole by hole:\s*([0-9]+(?:\/[0-9]+)*)\s*\(\s*par\s*([0-9]+(?:\/[0-9]+)*)/i);
  if (!m) return null;
  var s = m[1].split("/").map(function (x) { return parseInt(x, 10); });
  var p = m[2].split("/").map(function (x) { return parseInt(x, 10); });
  if (!s.length || !p.length || s.length !== p.length) return null;
  for (var i = 0; i < s.length; i++) { if (isNaN(s[i]) || isNaN(p[i])) return null; }
  var holes = null;
  var hm = String(r.notes).match(/\[holes\s+([0-9]+(?:[ ,]+[0-9]+)*)\]/i);
  if (hm) {
    holes = hm[1].split(/[ ,]+/).map(function (x) { return parseInt(x, 10); });
    if (holes.length !== s.length) holes = null;
  }
  return { scores: s, pars: p, holes: holes };
}
function sum(a) { var t = 0; for (var i = 0; i < a.length; i++) t += (a[i] || 0); return t; }
function sumSlice(a, from, to) { var t = 0; for (var i = from; i < to && i < a.length; i++) t += (a[i] || 0); return t; }

function getData() {
  try {
    var d = call(ROUNDS);
    if (d && d.ok && d.stats) { writeJSON(path(CACHE), d); return { data: d, stale: false }; }
  } catch (e) {}
  var c = readJSON(path(CACHE));
  if (c && c.stats) return { data: c, stale: true };
  return { data: null, stale: false };
}

function wp(st, str, size, color, bold, center) {
  var t = st.addText(String(str));
  t.font = bold ? Font.boldSystemFont(size) : Font.systemFont(size);
  t.textColor = color;
  t.lineLimit = 1;
  t.minimumScaleFactor = 0.7;
  if (center) t.centerAlignText();
  return t;
}
function chip(st, label, fg, bg, size, url, pad) {
  var c = st.addStack();
  c.backgroundColor = bg;
  c.cornerRadius = 5;
  c.setPadding(2, pad || 5, 2, pad || 5);
  c.centerAlignContent();
  var t = wp(c, label, size, fg, true, true);
  if (url) c.url = url;
  return c;
}
function selfURL(extra) {
  return "scriptable:///run/" + encodeURIComponent(SELF) + (extra ? "?" + extra : "");
}

// The draft is the round in progress: what the widget edits and the card finishes.
function blankDraft(data) {
  var pars = [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4];
  var course = (data && data.stats && data.stats.lastCourse) || "Ala Wai";
  var island = "Oahu";
  var courses = (data && data.courses) ? data.courses : [];
  for (var i = 0; i < courses.length; i++) if (courses[i].name === course) island = courses[i].island || island;
  var rounds = (data && data.rounds) ? data.rounds : [];
  for (var r = 0; r < rounds.length; r++) {
    var pn = parseNotes(rounds[r]);
    if (rounds[r].course === course && pn) { pars = pn.pars.concat(pars.slice(pn.pars.length)); break; }
  }
  return { course: course, island: island, holes: 18, pars: pars, scores: [], preview: true };
}

  var preview = !!draft.preview && !(draft.scores && draft.scores.length);

  var courseName = draft.course;
  if (preview && rounds.length && rounds[0].course) {
    var r0 = rounds[0];
    var pn = parseNotes(r0);
    var hs = (r0.holeScores && r0.holeScores.length) ? r0.holeScores : (pn ? pn.scores : null);
    if (hs && hs.length) {

    }

  if (hole < 1) hole = 1;
  if (hole > 18) hole = 18;

  var par = draft.pars[idx] || 4;
  var score = (shownScores[idx] === undefined || shownScores[idx] === null) ? null : shownScores[idx];

  var tot = 0, parPlayed = 0, played = 0;
  for (var i = 0; i < 18; i++) {

  var ct = wp(r2, courseName || "Pick a course", 11, ink, true);
  ct.lineLimit = 1; ct.minimumScaleFactor = 0.6;
  r2.addSpacer();
  wp(r2, played ? (tot + "  " + sign(tot - parPlayed)) : "no scores yet", 11, played ? green : soft, true);
  w.addSpacer(6);

  // row 3: hole + par
  var r3 = w.addStack(); r3.layoutHorizontally(); r3.centerAlignContent();
  wp(r3, "HOLE " + hole + " OF 18", 9, soft, true);
  r3.addSpacer();
  wp(r3, "PAR " + par, 9, soft, true);
  w.addSpacer(3);

  // row 4: the hole, with its arrows
  var big = w.addStack(); big.layoutHorizontally(); big.centerAlignContent();

  minus.backgroundColor = softBtn; minus.cornerRadius = 12;
  minus.size = new Size(52, 42); minus.centerAlignContent();
  wp(minus, "-", 26, ink, true, true);
  minus.url = selfURL("bump=" + hole + "&d=-1&hole=" + hole);
  big.addSpacer();
  var mid = big.addStack(); mid.layoutVertically(); mid.centerAlignContent();
  wp(mid, score === null ? "-" : String(score), 40, score === null ? soft : ink, true, true);
  wp(mid, score === null ? "tap to score" : (score === par ? "even par" : (score < par ? (par - score) + " under" : (score - par) + " over")), 9, soft, false, true);
  big.addSpacer();
  var plus = big.addStack();
  plus.backgroundColor = green; plus.cornerRadius = 12;
  plus.size = new Size(52, 42); plus.centerAlignContent();
  wp(plus, "+", 26, Color.white(), true, true);
  plus.url = selfURL("bump=" + hole + "&d=1&hole=" + hole);

  var r5 = w.addStack(); r5.layoutHorizontally(); r5.centerAlignContent(); r5.spacing = 6;
  chip(r5, "<", ink, softBtn, 15, selfURL("hole=" + (hole > 1 ? hole - 1 : 18)), 10);

  r5.addSpacer();
  chip(r5, "SAVE", Color.white(), green, 9, selfURL("save=1"), 9);
  chip(r5, "FULL CARD", ink, gold, 9, selfURL("card=1"), 9);
  w.addSpacer(5);

  // row 6: island progress + next course
  var r6 = w.addStack(); r6.layoutHorizontally(); r6.centerAlignContent();
  var ip = (data && data.islandProgress) ? data.islandProgress : null;

  if (ip) {

      if (ip[keys[z]].played < ip[keys[z]].total) { islandTxt = keys[z].toUpperCase() + " " + ip[keys[z]].played + "/" + ip[keys[z]].total; break; }
    }
  }
  if (islandTxt) wp(r6, islandTxt, 8, green, true);

  var nextName = "";

  var playedList = (data && data.coursesPlayedList) ? data.coursesPlayedList : [];
  for (var n = 0; n < courses.length; n++) if (playedList.indexOf(courses[n].name) === -1) { nextName = courses[n].name; break; }
  wp(r6, nextName ? ("NEXT " + nextName) : "ALL 73 PLAYED", 8, soft, false);

  w.url = selfURL("card=1");
  w.refreshAfterDate = new Date(Date.now() + 60 * 1000);
  return w;
}

function tap(act, h) {
  if (act === "par") {
    var order = [3, 4, 5, 6];
    var at = order.indexOf(parOf(h));
    pars[h] = order[(at + 1) % order.length];
  } else if (act === "up") {
    scores[h] = has(h) ? scores[h] + 1 : parOf(h);
  } else if (act === "down") {
    if (!has(h)) scores[h] = parOf(h);
    else if (scores[h] > 1) scores[h] = scores[h] - 1;
  }
  render();
}

function wipDraft() {
  var d = readJSON(path(DRAFT));
  if (!d || !d.course) { var got = getData(); d = blankDraft(got.data); }
  if (!d.scores) d.scores = [];
  if (!d.pars) d.pars = [4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4,4];
  return d;
}

function applyBump(hole, delta, page, keepHole) {
  var d = wipDraft();
  if (d.preview) { d.scores = []; d.preview = false; }
  var i = Number(hole) - 1;
  if (isNaN(i) || i < 0 || i > 17) return d;
  var par = d.pars[i] || 4;
  var has = !(d.scores[i] === undefined || d.scores[i] === null);
  var next = has ? d.scores[i] + Number(delta || 1) : par;
  if (next < 1) next = 1;
  if (next > 15) next = 15;
  d.scores[i] = next;
  if (page) d.page = Number(page);
  if (keepHole) d.hole = Math.min(18, Math.max(1, Number(keepHole)));
  writeJSON(path(DRAFT), d);
  return d;
}

function saveFromWidget() {
  var d = wipDraft();
  var scores = d.scores || [];
  var pars = d.pars || [];
  var used = [], usedPars = [], usedHoles = [];
  for (var hi = 0; hi < 18; hi++) {
    var sv = scores[hi];
    if (sv === undefined || sv === null) continue;
    used.push(sv);
    usedPars.push(pars[hi] || 4);
    usedHoles.push(hi + 1);
  }
  if (!used.length) return Script.complete();
  var body = {
    action: "create",
    course: d.course,
    island: d.island || "Other",
    holeScores: used,
    date: today(),
    score: sum(used),
    parsUsed: sum(usedPars),
    holesPlayed: used.length,
    notes: "Hole by hole: " + used.join("/") + " (par " + usedPars.join("/") + ") [holes " + usedHoles.join(",") + "]"
  };
  try { call(ROUNDS, body); } catch (e) {}
  d.scores = [];
  d.holes = 18;
  d.preview = false;
  writeJSON(path(DRAFT), d);
  dropFile(path(CACHE));
  var a = new Alert();
  a.title = "Round saved";
  a.message = body.holeScores.length + " holes at " + d.course + " sent to the Daily Dashboard.";
  a.addAction("Done");
  return a.present().then(function () { Script.complete(); });
}
