function fm() { return FileManager.local(); }
function path(name) { return fm().joinPath(fm().documentsDirectory(), name); }
function readJSON(p) { try { if (!fm().fileExists(p)) return null; return JSON.parse(fm().readString(p)); } catch (e) { return null; } }
function writeJSON(p, o) { try { fm().writeString(p, JSON.stringify(o)); } catch (e) {} }
function dropFile(p) { try { if (fm().fileExists(p)) fm().remove(p); } catch (e) {} }

function call(url, payload) {
  var req = new Request(url);
  req.timeoutInterval = 20;
  req.headers = { "Accept": "application/json" };
  if (payload) {
    req.method = "POST";
    req.headers["Content-Type"] = "application/json";
    req.body = JSON.stringify(payload);
  }
  return req.loadJSON();
}

function sign(n) {
  if (n === null || n === undefined || isNaN(n)) return "-";
  if (n === 0) return "E";
  return (n > 0 ? "+" : "") + n;
}
function today() {
  var d = new Date(), m = d.getMonth() + 1, day = d.getDate();
  return d.getFullYear() + "-" + (m < 10 ? "0" + m : m) + "-" + (day < 10 ? "0" + day : day);
}
function shortDate(iso) {
  if (!iso) return "";
  var p = String(iso).split("-");
  if (p.length !== 3) return String(iso);
  var M = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  return (M[parseInt(p[1], 10) - 1] || p[1]) + " " + parseInt(p[2], 10);
}
// Pull the hole-by-hole scores and pars out of a saved round's notes.
function parseNotes(r) {
  if (!r || !r.notes) return null;
  var m = String(r.notes).match(/Hole by hole:\s*([0-9]+(?:\/[0-9]+)*)\s*\(\s*par\s*([0-9]+(?:\/[0-9]+)*)/i);
  if (!m) return null;
  var s = m[1].split("/").map(function (x) { return parseInt(x, 10); });
  var p = m[2].split("/").map(function (x) { return parseInt(x, 10); });
  if (!s.length || !p.length || s.length !== p.length) return null;
  for (var i = 0; i < s.length; i++) { if (isNaN(s[i]) || isNaN(p[i])) return null; }
  return { scores: s, pars: p };
}
function sum(a) { var t = 0; for (var i = 0; i < a.length; i++) t += (a[i] || 0); return t; }
function sumSlice(a, from, to) { var t = 0; for (var i = from; i < to && i < a.length; i++) t += (a[i] || 0); return t; }

function getData() {
  try {
    var d = call(ROUNDS);
    if (d && d.ok && d.stats) { writeJSON(path(CACHE), d); return { data: d, stale: false }; }
  } catch (e) {}
  var c = readJSON(path(CACHE));
  if (c && c.stats) return { data: c, stale: true };
  return { data: null, stale: false };
}

function drawCell(dc, x, y, w, h, text, fg, bg, size, bold, align) {
  if (bg) { dc.setFillColor(bg); dc.fillRect(new Rect(x, y, w, h)); }
  dc.setFont(bold ? Font.boldSystemFont(size) : Font.systemFont(size));
  dc.setTextColor(fg);
  dc.setTextAlign(align || "center");
  var ty = y + (h - size) / 2 - 1;
  dc.drawTextInRect(String(text), new Rect(x, ty, w, h));
}

function deltaColor(d) {
  if (d < 0) return new Color("#7ddc9a");
  if (d > 0) return new Color("#ff8f7d");
  return new Color("#d8d8d8");
}

function drawNine(dc, y, first, count, pars, scores, W, tails) {
  var pad = 9;
  tails = tails || [];
  var cols = count + tails.length;
  var cw = (W - pad * 2) / cols;
  var x = pad;
  var rowHole = 13, rowPar = 15, rowScore = 17;
  var holeBg = new Color("#1f6f43"), parBg = new Color("#f2c94c");
  var parFg = new Color("#1a1a1a");
  var gridBg = new Color("#141618");
  var tailBg = new Color("#1c2420");

  for (var i = 0; i < count; i++) {
    drawCell(dc, x + i * cw, y, cw, rowHole, first + i, new Color("#eaf3ec"), holeBg, 8, true);
  }
  for (var t = 0; t < tails.length; t++) {
    drawCell(dc, x + (count + t) * cw, y, cw, rowHole, tails[t].label, new Color("#dff0e4"), holeBg, 8, true);
  }

  for (var j = 0; j < count; j++) {
    var pv = pars ? pars[first - 1 + j] : null;
    drawCell(dc, x + j * cw, y + rowHole, cw, rowPar, pv === null ? "-" : pv, parFg, parBg, 9, true);
  }
  for (var t2 = 0; t2 < tails.length; t2++) {
    drawCell(dc, x + (count + t2) * cw, y + rowHole, cw, rowPar, tails[t2].p ? tails[t2].p : "-", parFg, parBg, 9, true);
  }

  for (var k = 0; k < count; k++) {
    var idx = first - 1 + k;
    var sv = scores ? scores[idx] : null;
    var pvv = pars ? pars[idx] : null;
    var hasScore = !(sv === null || sv === undefined);
    var show = !hasScore ? "-" : (pvv === null || pvv === undefined ? String(sv) : sign(sv - pvv));
    var col = !hasScore ? new Color("#4a4a4a") : ((pvv === null || pvv === undefined) ? new Color("#d8d8d8") : deltaColor(sv - pvv));
    drawCell(dc, x + k * cw, y + rowHole + rowPar, cw, rowScore, show, col, gridBg, 10, true);
  }
  for (var t3 = 0; t3 < tails.length; t3++) {
    var tt = tails[t3];
    var tTxt = !tt.any ? "-" : (tt.p ? sign(tt.s - tt.p) : String(tt.s));
    var tCol = !tt.any ? new Color("#4a4a4a") : (tt.p ? deltaColor(tt.s - tt.p) : new Color("#d8d8d8"));
    drawCell(dc, x + (count + t3) * cw, y + rowHole + rowPar, cw, rowScore, tTxt, tCol, tailBg, 10, true);
  }
  return y + rowHole + rowPar + rowScore;
}

function tally(pars, scores, from, to) {
  var s = 0, p = 0, any = false;
  for (var i = from; i < to; i++) {
    if (pars && pars[i] !== undefined) p += pars[i];
    if (scores && scores[i] !== undefined) { s += scores[i]; any = true; }
  }
  return { s: s, p: p, any: any };
}

function avgToPar(rounds, holeCount) {
  var list = (rounds || []).filter(function (r) { return (r.holes || 18) === holeCount && typeof r.toPar === "number"; });
  if (!list.length) return null;
  var t = 0;
  for (var i = 0; i < list.length; i++) t += list[i].toPar;
  return Math.round(t / list.length);
}

function buildCard(data, draft) {
  var W = 345, H = 345;
  var dc = new DrawContext();
  dc.size = new Size(W, H);
  dc.opaque = true;
  dc.respectScreenScale = true;
  dc.setFillColor(new Color("#0b0b0d"));
  dc.fillRect(new Rect(0, 0, W, H));

  var gold = new Color("#d4af37"), dim = new Color("#8a8a8a"), white = new Color("#f2f0ea");

  // header
  drawCell(dc, 10, 8, 150, 14, "HAWAII GOLF", gold, null, 10, true, "left");
  var st = data ? data.stats : null;
  var hdr = "";
  if (st) {
    hdr = st.coursesPlayed + "/" + st.coursesTotal + " HI";
    if (st.worldPlayed) hdr += "  +" + st.worldPlayed + " world";
  }
  drawCell(dc, W - 165, 8, 155, 14, hdr, dim, null, 9, false, "right");

  var rounds = (data && data.rounds) ? data.rounds : [];

  function detailOf(r) {
    if (!r) return null;
    var pn = parseNotes(r);
    var sc = (r.holeScores && r.holeScores.length) ? r.holeScores : (pn ? pn.scores : null);
    if (!sc) return null;
    return { scores: sc, pars: pn ? pn.pars : null };
  }

  var detail = null, round = null;
  if (draft) {
    round = draft;
    detail = { pars: draft.pars, scores: draft.scores };
  } else {
    for (var i = 0; i < rounds.length; i++) {
      var dd = detailOf(rounds[i]);
      if (dd) { detail = dd; round = rounds[i]; break; }
    }
    if (!round && rounds.length) round = rounds[0];
  }

  if (!round) {
    drawCell(dc, 12, 140, W - 24, 20, "No rounds yet", dim, null, 12, false, "center");
    drawCell(dc, 12, 166, W - 24, 16, "Tap to start your first card", gold, null, 10, true, "center");
    return dc.getImage();
  }

  var pars = detail ? detail.pars : null;
  var scores = detail ? detail.scores : null;
  var played = round.holesPlayed || round.holes || (scores ? scores.length : 18);
  var parUsed = round.parsUsed || round.par || (pars ? tally(pars, null, 0, played).p : 0);
  var total = (round.score !== undefined && round.score !== null) ? round.score : (scores ? tally(pars, scores, 0, played).s : 0);
  var toPar = (typeof round.toPar === "number") ? round.toPar : (total - parUsed);

  drawCell(dc, 10, 24, 225, 18, round.course || "", white, null, 14, true, "left");
  drawCell(dc, W - 118, 25, 108, 16, (draft ? "IN PLAY" : shortDate(round.date)), draft ? gold : dim, null, 9, false, "right");
  drawCell(dc, 10, 42, 64, 26, String(total), white, null, 22, true, "left");
  drawCell(dc, 68, 46, 66, 20, sign(toPar), deltaColor(toPar), null, 13, true, "left");
  drawCell(dc, 136, 48, 198, 18, played + " holes  ·  par " + parUsed, dim, null, 9, false, "left");

  var y = 74;
  var front = Math.min(9, played);
  var frontTally = tally(pars, scores, 0, front);
  var frontTail = played > 9
    ? [{ label: "OUT", s: frontTally.s, p: frontTally.p, any: frontTally.any }]
    : [{ label: "TOTAL", s: frontTally.s, p: frontTally.p, any: frontTally.any }];
  y = drawNine(dc, y, 1, front, pars, scores, W, frontTail) + 5;
  if (played > 9) {
    var back = played - 9;
    var backTally = tally(pars, scores, 9, played);
    var allTally = tally(pars, scores, 0, played);
    y = drawNine(dc, y, 10, back, pars, scores, W, [
      { label: "IN", s: backTally.s, p: backTally.p, any: backTally.any },
      { label: "TOT", s: allTally.s, p: allTally.p, any: allTally.any }
    ]) + 5;
  }

  if (!detail) {
    drawCell(dc, 10, y + 2, W - 20, 14, "no hole detail on this round, tap to enter a new card", dim, null, 8, false, "center");
    y += 18;
  }

  // ---- what is still out there: island progress + next course + recent form
  var isl = (data && data.islandProgress) ? data.islandProgress : null;
  var curIsl = null;
  if (isl) {
    var order = ["Oahu", "Maui", "Kauai", "Hawaii", "Molokai", "Lanai"];
    for (var oi = 0; oi < order.length; oi++) {
      var ip = isl[order[oi]];
      if (ip && ip.played < ip.total) { curIsl = { name: order[oi], played: ip.played, total: ip.total }; break; }
    }
  }
  var yb = y + 10;
  if (curIsl) {
    drawCell(dc, 10, yb, 74, 12, curIsl.name.toUpperCase(), gold, null, 8, true, "left");
    var bx = 80, bw = W - 80 - 52, bh = 7, by = yb + 3;
    dc.setFillColor(new Color("#23282b"));
    dc.fillRect(new Rect(bx, by, bw, bh));
    var frac = curIsl.total ? curIsl.played / curIsl.total : 0;
    var fw = Math.max(frac > 0 ? 3 : 0, bw * frac);
    dc.setFillColor(new Color("#3f7d4f"));
    dc.fillRect(new Rect(bx, by, fw, bh));
    drawCell(dc, W - 50, yb, 40, 12, curIsl.played + "/" + curIsl.total, new Color("#cfd6d9"), null, 8, false, "right");
    yb += 16;
  }

  var nextName = "";
  if (data && data.courses && data.coursesPlayedList) {
    for (var ci = 0; ci < data.courses.length; ci++) {
      if (data.coursesPlayedList.indexOf(data.courses[ci].name) === -1) { nextName = data.courses[ci].name + " · " + (data.courses[ci].island || ""); break; }
    }
  }
  if (nextName) {
    drawCell(dc, 10, yb, 44, 12, "NEXT", new Color("#6b7175"), null, 8, true, "left");
    drawCell(dc, 54, yb, W - 64, 12, nextName, gold, null, 9, true, "left");
    yb += 18;
  }

  var recent = [];
  for (var ri = 0; ri < rounds.length && recent.length < 4; ri++) recent.push(rounds[ri]);
  if (recent.length) {
    drawCell(dc, 10, yb, 120, 10, "RECENT", new Color("#6b7175"), null, 7, true, "left");
    yb += 12;
    for (var rj = 0; rj < recent.length; rj++) {
      var rr = recent[rj];
      drawCell(dc, 10, yb, 56, 12, shortDate(rr.date), dim, null, 8, false, "left");
      drawCell(dc, 66, yb, 200, 12, rr.course || "", new Color("#cfd6d9"), null, 8, false, "left");
      var tp = (typeof rr.toPar === "number") ? rr.toPar : null;
      drawCell(dc, W - 70, yb, 60, 12, String(rr.score) + (tp === null ? "" : " " + sign(tp)), tp === null ? white : deltaColor(tp), null, 8, true, "right");
      yb += 13;
    }
  }

  // footer
  if (st) {
    var a18 = avgToPar(rounds, 18), a9 = avgToPar(rounds, 9);
    var avgTxt = a18 !== null ? ("AVG18 " + sign(a18)) : (a9 !== null ? ("AVG9 " + sign(a9)) : "AVG -");
    drawCell(dc, 10, H - 42, 100, 14, "ROUNDS " + st.totalRounds, white, null, 9, true, "left");
    drawCell(dc, 118, H - 42, 96, 14, "BEST " + sign(st.bestToPar), white, null, 9, true, "left");
    drawCell(dc, 220, H - 42, 115, 14, avgTxt, white, null, 9, true, "left");
  }
  drawCell(dc, 10, H - 26, W - 90, 14, "TAP TO EDIT THE CARD", gold, null, 9, true, "left");
  drawCell(dc, W - 80, H - 26, 70, 14, "play ▸", dim, null, 9, false, "right");
  return dc.getImage();
}

function renderWidget() {
  var w = new ListWidget();
  w.backgroundColor = new Color("#0b0b0d");
  w.setPadding(0, 0, 0, 0);
  var got = getData();
  var draft = readJSON(path(DRAFT));
  try {
    w.addImage(buildCard(got.data, draft));
  } catch (e) {
    var t = w.addText("HAWAII GOLF\ndraw error: " + String(e).slice(0, 60));
    t.font = Font.systemFont(9);
    t.textColor = Color.white();
  }
  w.url = "scriptable:///run/" + encodeURIComponent(SELF);
  w.refreshAfterDate = new Date(Date.now() + 15 * 60 * 1000);
  Script.setWidget(w);
  Script.complete();
}
